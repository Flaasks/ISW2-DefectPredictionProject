public class BookkeeperMessages {

    private static final Map<Integer, String> CODE_MESSAGES = Map.ofEntries(
        Map.entry(Code.OK, "No problem"),
        Map.entry(Code.ReadException, "Error while reading ledger"),
        Map.entry(Code.QuorumException, "Invalid quorum size on ensemble size"),
        Map.entry(Code.NoBookieAvailableException, "Invalid quorum size on ensemble size"),
        Map.entry(Code.DigestNotInitializedException, "Digest engine not initialized"),
        Map.entry(Code.DigestMatchException, "Entry digest does not match"),
        Map.entry(Code.NotEnoughBookiesException, "Not enough non-faulty bookies available"),
        Map.entry(Code.NoSuchLedgerExistsException, "No such ledger exists"),
        Map.entry(Code.BookieHandleNotAvailableException, "Bookie handle is not available"),
        Map.entry(Code.ZKException, "Error while using ZooKeeper"),
        Map.entry(Code.MetaStoreException, "Error while using MetaStore"),
        Map.entry(Code.LedgerRecoveryException, "Error while recovering ledger"),
        Map.entry(Code.LedgerClosedException, "Attempt to write to a closed ledger"),
        Map.entry(Code.WriteException, "Write failed on bookie"),
        Map.entry(Code.NoSuchEntryException, "No such entry"),
        Map.entry(Code.IncorrectParameterException, "Incorrect parameter input"),
        Map.entry(Code.InterruptedException, "Interrupted while waiting for permit"),
        Map.entry(Code.ProtocolVersionException, "Bookie protocol version on server is incompatible with client"),
        Map.entry(Code.MetadataVersionException, "Bad ledger metadata version"),
        Map.entry(Code.LedgerFencedException, "Ledger has been fenced off. Some other client must have opened it to read"),
        Map.entry(Code.UnauthorizedAccessException, "Attempted to access ledger using the wrong password"),
        Map.entry(Code.UnclosedFragmentException, "Attempting to use an unclosed fragment; This is not safe"),
        Map.entry(Code.WriteOnReadOnlyBookieException, "Attempting to write on ReadOnly bookie")
    );
    
    public static String getMessage(int code) {
        return CODE_MESSAGES.getOrDefault(code, "Invalid operation");
    }

    private String lastMessage; 

    public BookkeeperMessages(int code) {
    this.lastMessage = CODE_MESSAGES.getOrDefault(code, "Invalid operation");
    }

}
